! ===== R1 - Router (router-on-a-stick, DHCP, ACL, NAT, SSH) =====
! Lab use only. Change every password before using anywhere else.
enable
configure terminal
hostname R1
no ip domain-lookup
service password-encryption
enable secret LabOnly#2026
ip domain-name lab.local
username admin secret LabOnly#2026
crypto key generate rsa modulus 2048
ip ssh version 2

banner motd # Authorized access only #

! --- DHCP ---
ip dhcp excluded-address 192.168.10.1 192.168.10.10
ip dhcp excluded-address 192.168.20.1 192.168.20.10
ip dhcp pool STAFF
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 8.8.8.8
ip dhcp pool GUEST
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 8.8.8.8

! --- Inter-VLAN routing (trunk from SW1) ---
interface GigabitEthernet0/0
 description TRUNK-to-SW1
 no shutdown
interface GigabitEthernet0/0.10
 description STAFF
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip nat inside
interface GigabitEthernet0/0.20
 description GUEST
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip access-group GUEST_IN in
 ip nat inside
interface GigabitEthernet0/0.99
 description MGMT
 encapsulation dot1Q 99
 ip address 192.168.99.1 255.255.255.0
interface GigabitEthernet0/0.999
 description NATIVE-UNUSED
 encapsulation dot1Q 999 native

! --- Guest isolation ---
ip access-list extended GUEST_IN
 deny ip 192.168.20.0 0.0.0.255 192.168.10.0 0.0.0.255
 deny ip 192.168.20.0 0.0.0.255 192.168.99.0 0.0.0.255
 permit ip any any

! --- WAN + NAT (203.0.113.0/30 is a documentation range) ---
interface GigabitEthernet0/1
 description WAN-to-ISP
 ip address 203.0.113.2 255.255.255.252
 ip nat outside
 no shutdown
access-list 1 permit 192.168.10.0 0.0.0.255
access-list 1 permit 192.168.20.0 0.0.0.255
ip nat inside source list 1 interface GigabitEthernet0/1 overload
ip route 0.0.0.0 0.0.0.0 203.0.113.1

! --- Management: SSH only, from MGMT VLAN only ---
access-list 10 permit 192.168.99.0 0.0.0.255
line vty 0 4
 access-class 10 in
 login local
 transport input ssh
line console 0
 logging synchronous
end
write memory
